\begin{tikzpicture}[x=1cm,y=1cm,font=\footnotesize,
      nd/.style={rounded corners=2pt,line width=0.8pt,draw=ink!55,fill=white,align=center,inner sep=3pt,font=\footnotesize,nohyph},
      fw/.style={rounded corners=2pt,line width=1.1pt,draw=primary,fill=primary!7,align=center,inner sep=3pt,font=\footnotesize,nohyph},
      ar/.style={-{Stealth[length=2.0mm]},line width=0.9pt,draw=ink!70},
      la/.style={font=\footnotesize,text=primary,inner sep=1pt}]
    \path (-0.2,-1.0) rectangle (15.3,6.1);
    % ================= step 1: what is solved, and why mu
    \node[anchor=north west,font=\small,align=left,inner sep=0pt] at (0.0,5.95)
      {$\vgh=\arg\min_{\vg}\ \|\mathbf A\vg-\sv\|^2+\textcolor{amber!85!black}{\mu^2\|\vg\|^2}$
       \quad$\Longleftrightarrow$\quad $(\mathbf A^H\mathbf A+\mu^2\mathbf I)\,\vgh=\mathbf A^H\sv$};
    \node[anchor=north west,font=\footnotesize,text=amber!85!black,align=left,inner sep=0pt] at (0.0,5.2)
      {$\mu^2\|\vg\|^2$: keeps small what the channels do not see (outside the echo band)};
    % ================= step 2: P4, the size of mu, before the loop
    \begin{scope}[visible on=<2->]
      \node[nd,text width=3.85cm,anchor=north west] (p4) at (0.0,4.35)
        {\textbf{before the loop}: $\sigma_{\max}(\mathbf A)$\\ by power iterations:\\[0.3ex]
         $\mathbf x\leftarrow\mathbf A^H(\mathbf A\,\mathbf x)/\|\cdot\|$,\enspace 12 times\\[0.3ex]
         $\sigma_{\max}=\sqrt{\|\mathbf A^H\mathbf A\,\mathbf x\|}$\enspace$\to\ \mu=0.03\,\sigma_{\max}$};
    \end{scope}
    % ================= step 3: one pass of the loop: one A and one A^H per iteration
    \begin{scope}[visible on=<3->]
      \fill[panel!55,rounded corners=4pt] (4.55,-0.4) rectangle (15.25,4.4);
      \draw[primary,rounded corners=4pt,line width=0.7pt] (4.55,-0.4) rectangle (15.25,4.4);
      \node[anchor=north west,font=\small\bfseries,text=primary] at (4.65,4.35) {one pass: the iterative solver};
      \node[nd] (st) at (7.0,3.5) {start: $\mathbf r=\sv$,\enspace $\mathbf d=\mathbf 0$};
      \node[fw] (z0) at (11.6,3.5) {$\mathbf z=\mathbf A^H\mathbf r$,\enspace $\mathbf p=\mathbf z$};
      \node[la,anchor=south] at (z0.north) {backward};
      \draw[ar] (st) -- (z0);
      \node[fw,minimum height=0.95cm] (n1) at (6.35,2.15) {$\mathbf q=\mathbf A\,\mathbf p$};
      \node[la,anchor=east] at (n1.west) {forward};
      \node[nd,minimum height=0.95cm] (n2) at (9.45,2.15) {$\alpha=\dfrac{\|\mathbf z\|^2}{\|\mathbf q\|^2+\mu^2\|\mathbf p\|^2}$\\[0.2ex]
        $\mathbf d\mathrel{+}=\alpha\,\mathbf p$,\enspace $\mathbf r\mathrel{-}=\alpha\,\mathbf q$};
      \node[fw,minimum height=0.95cm] (n3) at (13.25,2.15) {$\mathbf z=\mathbf A^H\mathbf r-\mu^2\,\mathbf d$};
      \node[la,anchor=south] at (n3.north) {backward};
      
      \node[nd,draw=amber,line width=0.9pt] (n4) at (13.25,0.75) {$\|\mathbf z\|\le10^{-4}\,\|\mathbf z_0\|$\\ or 60 iterations?};
      \node[nd] (n5) at (6.35,0.75) {$\mathbf p=\mathbf z+\beta\,\mathbf p$\\ {\color{muted}$\beta=\|\mathbf z\|^2/\|\mathbf z_{\mathrm{prev}}\|^2$}};
      \draw[ar] (z0.south) |- ($(n1.north)+(0,0.35)$) -- (n1.north);
      \draw[ar] (n1) -- (n2); \draw[ar] (n2) -- (n3); \draw[ar] (n3.south) -- ++(0,-0.38) -- (n4.north);
      \draw[ar] (n4) -- node[above,font=\footnotesize]{no} (n5);
      \draw[ar,draw=amber] (n5.north) -- node[left,font=\footnotesize,text=amber!85!black]{repeat} (n1.south);
      \draw[ar] (n4.south) -- (13.25,-0.12) node[right,font=\footnotesize,inner sep=2pt]{yes: return $\mathbf d$};
      \node[anchor=west,font=\footnotesize,text=ink!80,align=left] at (4.65,-0.12)
        {each iteration: one $\mathbf A$ and one $\mathbf A^H$, plus sums of vectors};
    \end{scope}
    % ================= step 4: the second pass (iterated Tikhonov)
    \begin{scope}[visible on=<4->]
      \node[nd,text width=3.85cm,anchor=north west,draw=cjoint,line width=0.9pt] (p2) at (0.0,1.6)
        {\textbf{2 passes}: pass 1 gives $\vgh_1$;\\ pass 2 solves the same\\ way for what is left,\\
         $\mathbf r=\sv-\mathbf A\vgh_1$;\enspace $\vgh=\vgh_1+\mathbf d_2$};
      \node[anchor=north,font=\footnotesize,text=muted,align=center] at (p2.south) {iterated Tikhonov: undoes most\\ of the damping $\mu$ adds};
    \end{scope}
  \end{tikzpicture}
