\chapter{Joint topography correction and reconstruction}\label{ch:joint}

The phase $\dC$ of every ground position is known from the DEM, but it cannot be removed from each channel before the
reconstruction: a Doppler bin mixes parcels from positions $X\approx\SI{6.2}{km}$ apart that need different phases
(\cref{fig:binconflict}). The joint reconstruction separates the parcels and applies their phases in one system.

\section{From the DEM to the phase of every channel}\label{sec:dem-phase}

$\Delta C_{0,i}$ depends on height and channel, not on azimuth, so it is computed once per channel at 33 heights between the
DEM minimum and maximum (exact range histories of a virtual point on the iso-range surface) and fitted by a cubic. The DEM
then gives the phase map
\begin{equation}
  \varphi_i(x)=\frac{2\pi}{\lambda}\,\Delta C_{0,i}\bigl(h(x)\bigr).
  \label{eq:phasemap}
\end{equation}
It is the only terrain information the reconstruction gets.

\section{The model and its solution}\label{sec:joint-system}

Unknowns are red, data blue.

\subsection{Building the model}\label{sec:joint-build}

\noindent Samples $n$ at $\PRF=\SI{2000}{Hz}$, $N=12288$; $\mathcal F$ is the DFT; $k[n]$ is the echo of one point (azimuth
chirp), $K(f)=\mathcal F\{k\}$.

\case{1} One target at the reference point, monostatic, sampled at $\PRF$:
\begin{equation*}
  u[n]=k[n]\ \xrightarrow{\ \mathcal F\ }\ U(f)=K(f).
\end{equation*}

\case{2} \caseref{1} with the target at $\ell=x/d$, $d=v/\PRF=\SI{3.844}{m}$:
\begin{equation*}
  u[n]=k[n-\ell]\ \xrightarrow{\ \mathcal F\ }\ U(f)=K(f)\,\e^{-\jj2\pi f\ell/\PRF}.
\end{equation*}

\case{3} \caseref{2} with several targets at $\ell_p$, with amplitudes $a_p$:
\begin{gather}
  u[n]=\sum_p a_p\,k[n-\ell_p]\ \xrightarrow{\ \mathcal F\ }\ U(f)=\sum_p a_pK(f)\,\e^{-\jj2\pi f\ell_p/\PRF}
  =K(f)\underbrace{\Bigl[\sum_p a_p\,\e^{-\jj2\pi f\ell_p/\PRF}\Bigr]}_{G(f)=\mathcal F\{\gs\}},
  \notag\\
  \gs[\ell]=\sum_p a_p\,\delta[\ell-\ell_p]. \label{eq:case3}
\end{gather}
The focused scene $\gs$ (one peak per target) is the unknown.

\case{4} \caseref{1} with $\Nrx$ channels, each a filter on the monostatic echo:
\begin{equation*}
  K_i(f):=H_i(f)\,K(f),\qquad H_i(f)=\e^{-\jj2\pi\left[C_{0,i}/\lambda+(\Delta t_i-C_{1,i})f+C_{2,i}\lambda f^2\right]}
  \quad\text{\eqref{eq:Hi}}.
\end{equation*}
$K$ is simulated once (\cref{fig:kflow}).

\centering

\caption[Where $H_i$ and the kernel $K$ come from]{$H_i$ and $K$, both from the $h=0$ point. Top: the channel-$i$ range

\label{fig:kflow}
\end{figure}

\case{5} \caseref{3} $+$ \caseref{4}:
\begin{equation*}
  S_i(f)=K_i(f)\,G(f).
\end{equation*}

\case{6} \caseref{5} with relief: in channel $i$ each target gets the phase
$\varphi_{i,p}=\frac{2\pi}{\lambda}\Delta C_{0,i}(\ell_p)$ of \eqref{eq:dc0},
\begin{gather*}
  s_i[n]=\sum_p a_p\,\e^{-\jj\varphi_{i,p}}\,k_i[n-\ell_p]\ \xrightarrow{\ \mathcal F\ }\
  S_i(f)=K_i(f)\underbrace{\Bigl[\sum_p a_p\,\e^{-\jj\varphi_{i,p}}\,\e^{-\jj2\pi f\ell_p/\PRF}\Bigr]}_{\mathcal F\{\e^{-\jj\varphi_i}\gs\}}\\[2pt]
  \Longrightarrow\quad
  \boxed{\ S_i(f)=H_i(f)\,K(f)\,\mathcal F\{\theta_i\odot\gs\}(f),\qquad \theta_i[\ell]=\e^{-\jj\varphi_i[\ell]}\ }
\end{gather*}
The phase depends on position and channel, not on frequency: a pointwise \emph{screen} $\theta_i$ on $\gs$ before the
DFT. This is why the focused line is the unknown: there the relief is one multiplication per position, whereas in the raw
line each echo spreads over its aperture.

\case{7} \caseref{6} recorded by the channel, one sample in $\Nrx$ (\SI{2000}{Hz} $\to$ \SI{500}{Hz}):
\begin{equation}
  \sv_i[k]=s_i^{\mathrm{full}}[\Nrx k]
  \quad\Longrightarrow\quad
  \boxed{\ \sv_i=\underbrace{\mathcal D}_{\downarrow\Nrx}\ \underbrace{\mathbf F^{-1}}_{\text{IFFT}}\
  \underbrace{\mathrm{diag}(K_i)}_{\odot K_i}\ \underbrace{\mathbf F}_{\text{FFT}}\
  \underbrace{\mathrm{diag}(\theta_i)}_{\odot\,\e^{-\jj\varphi_i}}\ \vg=\mathbf A_i\vg\ }
  \label{eq:Ai}
\end{equation}
with $K_i=H_i\odot K$. Stacking the channels gives $\sv=\mathbf A\vg$: $12288$ equations ($4\times3072$) for $12288$
unknowns (\cref{fig:modelchain,fig:aproduct}; $\Psi_i=\mathrm{diag}(\theta_i)$).

\subsection{What is stored and what is applied}

The model and its adjoint are
\begin{equation}
  \mathbf A_i\vg=\mathcal D\,\mathbf F^{-1}\mathrm{diag}(K_i)\,\mathbf F\,\mathrm{diag}(\theta_i)\,\vg,
  \qquad
  \mathbf A\Hm\sv=\sum_i \mathrm{diag}(\theta_i)\Hm\,\mathbf F\Hm\,\mathrm{diag}(K_i)\Hm\,(\mathbf F^{-1})\Hm\,\mathcal D\Hm\,\sv_i ,
  \label{eq:adjoint}
\end{equation}
with $\mathbf F\Hm=N\mathbf F^{-1}$, $(\mathbf F^{-1})\Hm=\mathbf F/N$ and $\mathcal D\Hm$ the insertion of $\Nrx-1$ zeros
between samples: the adjoint focuses each channel, removes its relief phase and sums them, like a SAR processor.

Stored, $\mathbf A$ would take $12288^2\times\SI{16}{B}\approx\SI{2.4}{GB}$ and about $10^8$ operations per product. Every
factor is diagonal or a fast transform, so only the diagonals are kept (\cref{tab:operators}): about \SI{1.6}{MB} and eight
FFTs per product with four channels (\cref{fig:axeschain,fig:afill,fig:afunctions}).

Without relief $\mathbf A$ splits into $3072$ independent $4\times4$ blocks, the standard reconstruction \eqref{eq:fold};
relief mixes the bins, so the whole line is solved at once.

\subsection{The problem that is solved}

\begin{equation}
  \hat{\vg}=\arg\min_{\vg}\ \|\mathbf A\vg-\sv\|^2+\mu^2\|\vg\|^2
  \quad\Longleftrightarrow\quad
  \bigl(\mathbf A\Hm\mathbf A+\mu^2\mathbf I\bigr)\,\hat{\vg}=\mathbf A\Hm\sv ,
  \qquad \mu=0.03\,\sigma_{\max}(\mathbf A).
  \label{eq:tikhonov}
\end{equation}
$\sigma_{\max}$ comes from 12 power iterations:
\[
  \bm x\leftarrow\mathbf A\Hm\mathbf A\bm x/\|\mathbf A\Hm\mathbf A\bm x\|,\qquad
  \sigma_{\max}\approx\sqrt{\|\mathbf A\Hm\mathbf A\bm x\|}.
\]

\paragraph{Why regularise.} With $\mathbf A\bm v_k=\sigma_k\bm u_k$, each direction $\bm v_k$ gets its own weight:
\begin{equation}
  \hat{\vg}=\sum_k \underbrace{\phi_k}_{\text{weight}}\,(\bm u_k\Hm\sv)\,\bm v_k ,
  \qquad
  \phi_k=\begin{cases}
    1/\sigma_k & \text{without $\mu$ (inverse)},\\[2pt]
    \dfrac{\sigma_k}{\sigma_k^2+\mu^2} & \text{with $\mu$ (Tikhonov)}.
  \end{cases}
  \label{eq:filterfactors}
\end{equation}
A small $\sigma_k$ is a direction the channels hardly measure (e.g.\ parcels seen with almost the same phases), where
$1/\sigma_k$ would amplify noise and model errors. With $\mu=0.03\,\sigma_{\max}$, $\phi_k\approx1/\sigma_k$ for
$\sigma_k\gg\mu$ and only directions below about \SI{3}{\percent} of the best are damped.

\subsection{Conjugate gradients for least squares}

CGLS solves \eqref{eq:tikhonov} with products by $\mathbf A$ and $\mathbf A\Hm$ only, never forming the normal matrix
\cite{Hestenes1952,Trefethen1997} (\cref{fig:solveloop}):
\begin{itemize}
  \item each iteration: one model ($\bm q=\mathbf A\bm p$) and one adjoint ($\bm z=\mathbf A\Hm\bm r-\mu^2\bm\delta$);
        step $\alpha=\|\bm z\|^2/(\|\bm q\|^2+\mu^2\|\bm p\|^2)$, new direction $\bm p\leftarrow\bm z+\beta\bm p$,
        $\beta=\|\bm z_{\mathrm{new}}\|^2/\|\bm z\|^2$;
  \item a pass stops at $\|\bm z\|\le10^{-4}\|\bm z_0\|$ or 60 iterations;
  \item the second pass solves for the residual of the first and adds it (iterated Tikhonov); the weight becomes
        $\bigl[1-\bigl(\mu^2/(\sigma_k^2+\mu^2)\bigr)^2\bigr]/\sigma_k$, closer to $1/\sigma_k$ where the channels measure
        well and still zero where they do not.
\end{itemize}

\subsection{The answer: \texorpdfstring{$U=K\odot\mathbf F\hat{\vg}$}{U = K F g}}\label{sec:joint-answer}

The output is the line an ideal \emph{monostatic} radar at the transmitter would record at $\PRF=\SI{2000}{Hz}$ (the
reference), i.e.\ \caseref{3} applied to the scene found:
\begin{equation}
  U(f)=\underbrace{K(f)}_{\text{echo of one point}}\ \underbrace{\hat G(f)}_{\mathcal F\{\hat{\gs}\}}
  \quad\Longleftrightarrow\quad
  u[n]=\sum_\ell \hat{\gs}[\ell]\,k[n-\ell],
  \label{eq:answer}
\end{equation}
one chirp per position, weighted by $\hat{\gs}$.

No screen is applied: it is the phase relative to the monostatic path, so $\theta\equiv1$ for the monostatic radar
($\bat=\bxt=0$); the common monostatic phase of each target is part of $a_p$ and estimated in $\hat{\vg}$. Returning the raw
line ($12288$ samples, \SI{2000}{Hz}, $U(f)=0$ for $|f|>\Ba/2$, $\Ba=\SI{1234.8}{Hz}$) rather than $\hat{\vg}$ keeps the
comparison fair (every method is focused the same way), discards the part of $\vg$ outside the band that the data do not
determine, and fits the SAR chain, which expects raw data. Focusing,
$\text{image}=\mathbf F^{-1}\{K^*U\}=\mathbf F^{-1}\{|K|^2\hat G\}$, gives one peak per target ($\SI{-3}{dB}$ width
$\approx\SI{5.8}{m}$), scored with the metric of \cref{sec:metric} (\cref{fig:answer}).

\section{Properties}

\begin{itemize}
  \item \textbf{Consistency.} With a zero DEM it reproduces the standard reconstruction within \SI{0.1}{\percent} rms.
  \item \textbf{Information used.} Only the channels, the geometry and the DEM; removing the target positions from the
        configuration leaves the output bit-identical.
  \item \textbf{DEM self-test.} The unexplained part of the channels is \SIrange{-41}{-39}{dB} with the right DEM and
        \SIrange{-20}{-7}{dB} with a wrong one (mirrored or shifted by \SI{1}{km}).
  \item \textbf{Cost.} About \SI{3.3}{s} per line with four channels (median over 680 runs).
  \item \textbf{Stability.} Three or more channels with unequally spaced cross-track baselines make almost every direction
        well measured (\cref{sec:mismatch}).
\end{itemize}
