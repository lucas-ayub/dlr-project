\chapter{Joint topography correction and reconstruction}\label{ch:joint}

The phase $\dC$ of every ground position (\cref{sec:dc0}) is known from the DEM, but it cannot be removed from each channel
before the reconstruction: every Doppler bin mixes parcels from positions $X\approx\SI{6.2}{km}$ apart that need different
phases, and a channel correction gives the whole bin one phase (\cref{fig:binconflict}). Correcting the parcels needs them
separated, and separating them needs the phases corrected. The joint reconstruction does both in one system, with the DEM
phase inside the inversion.

\section{From the DEM to the phase of every channel}\label{sec:dem-phase}

$\Delta C_{0,i}$ depends on height and channel, not on azimuth (\cref{sec:dc0}), so it is computed once per channel: at 33
heights $h_q$ equally spaced between the DEM minimum and maximum, from the exact range histories of a virtual point on the
iso-range surface. A cubic fit gives $\Delta C_{0,i}(h)$, and the DEM height of every position $x$ gives the phase map
\begin{equation}
  \varphi_i(x)=\frac{2\pi}{\lambda}\,\Delta C_{0,i}\bigl(h(x)\bigr).
  \label{eq:phasemap}
\end{equation}
It is zero for the transmitter channel, constant for a flat DEM, and the only terrain information the reconstruction gets.

\section{The model and its solution}\label{sec:joint-system}

The channel model is built from the echo of one point, written as a linear system, solved without forming its matrix, and
turned into the full-rate line $U=K\odot\mathbf F\hat{\vg}$. Unknowns are red, data blue.

\subsection{Building the model}\label{sec:joint-build}

\noindent\emph{Full-rate line:} $n$ is a sample at $\PRF=\SI{2000}{Hz}$, $N=12288$; $\mathcal F$ is the DFT; $k[n]$ is the
echo of one point (the azimuth chirp) and $K(f)=\mathcal F\{k\}$.

\case{1} One target at the reference point, monostatic, sampled at $\PRF$:
\begin{equation*}
  u[n]=k[n]\ \xrightarrow{\ \mathcal F\ }\ U(f)=K(f).
\end{equation*}

\case{2} \caseref{1} with the target at $\ell=x/d$, $d=v/\PRF=\SI{3.844}{m}$:
\begin{equation*}
  u[n]=k[n-\ell]\ \xrightarrow{\ \mathcal F\ }\ U(f)=K(f)\,\e^{-\jj2\pi f\ell/\PRF}.
\end{equation*}

\case{3} \caseref{2} with several targets at $\ell_p$, with amplitudes $a_p$:
\begin{gather}
  u[n]=\sum_p a_p\,k[n-\ell_p]\ \xrightarrow{\ \mathcal F\ }\ U(f)=\sum_p a_pK(f)\,\e^{-\jj2\pi f\ell_p/\PRF}
  =K(f)\underbrace{\Bigl[\sum_p a_p\,\e^{-\jj2\pi f\ell_p/\PRF}\Bigr]}_{G(f)=\mathcal F\{\gs\}},
  \notag\\
  \gs[\ell]=\sum_p a_p\,\delta[\ell-\ell_p]. \label{eq:case3}
\end{gather}
$\gs$, one peak per target, is the focused scene: the unknown.

\case{4} \caseref{1} with $\Nrx$ channels: each is a filter on the monostatic echo,
\begin{equation*}
  K_i(f):=H_i(f)\,K(f),\qquad H_i(f)=\e^{-\jj2\pi\left[C_{0,i}/\lambda+(\Delta t_i-C_{1,i})f+C_{2,i}\lambda f^2\right]}
  \quad\text{\eqref{eq:Hi}}.
\end{equation*}
$K$ is the DFT of the echo of the $h=0$ point, simulated once by the monostatic radar at the full PRF; $H_i$ moves it to
receiver $i$ (\cref{fig:kflow}).

\case{5} \caseref{3} $+$ \caseref{4}:
\begin{equation*}
  S_i(f)=K_i(f)\,G(f).
\end{equation*}

\case{6} \caseref{5} with relief: in channel $i$ each target gets the phase
$\varphi_{i,p}=\frac{2\pi}{\lambda}\Delta C_{0,i}(\ell_p)$ of \eqref{eq:dc0},
\begin{gather*}
  s_i[n]=\sum_p a_p\,\e^{-\jj\varphi_{i,p}}\,k_i[n-\ell_p]\ \xrightarrow{\ \mathcal F\ }\
  S_i(f)=K_i(f)\underbrace{\Bigl[\sum_p a_p\,\e^{-\jj\varphi_{i,p}}\,\e^{-\jj2\pi f\ell_p/\PRF}\Bigr]}_{\mathcal F\{\e^{-\jj\varphi_i}\gs\}}\\[2pt]
  \Longrightarrow\quad
  \boxed{\ S_i(f)=H_i(f)\,K(f)\,\mathcal F\{\theta_i\odot\gs\}(f),\qquad \theta_i[\ell]=\e^{-\jj\varphi_i[\ell]}\ }
\end{gather*}
The phase depends on position $\ell$ and channel, not on frequency, so it is a pointwise \emph{screen} $\theta_i$ applied
to $\gs$ \emph{before} the DFT. This is why the focused line is the unknown: there the relief of each target is one
multiplication at its own position, whereas in the raw line each echo spreads over its aperture and mixes with the terrain $X$
away (\cref{fig:binconflict}).

\case{7} \caseref{6} recorded by the channel, which keeps one sample in $\Nrx$ of the full line (\SI{2000}{Hz}, $N=12288$ $\to$
\SI{500}{Hz}):
\begin{equation}
  \sv_i[k]=s_i^{\mathrm{full}}[\Nrx k]
  \quad\Longrightarrow\quad
  \boxed{\ \sv_i=\underbrace{\mathcal D}_{\downarrow\Nrx}\ \underbrace{\mathbf F^{-1}}_{\text{IFFT}}\
  \underbrace{\mathrm{diag}(K_i)}_{\odot K_i}\ \underbrace{\mathbf F}_{\text{FFT}}\
  \underbrace{\mathrm{diag}(\theta_i)}_{\odot\,\e^{-\jj\varphi_i}}\ \vg=\mathbf A_i\vg\ }
  \label{eq:Ai}
\end{equation}
with $K_i=H_i\odot K$. Stacking the channels: $\sv=\mathbf A\vg$, with $\sv\in\mathbb C^{4\times3072}$ ($12288$ equations) and
$\vg\in\mathbb C^{12288}$ ($12288$ unknowns).

\Cref{fig:modelchain} names each factor of \eqref{eq:Ai} ($\Psi_i=\mathrm{diag}(\theta_i)$); \cref{fig:aproduct} gives
their sizes and axes and the stacking of the blocks $\mathbf A_i$ into $\mathbf A$.

\subsection{What is stored and what is applied}

The model and its adjoint are
\begin{equation}
  \mathbf A_i\vg=\mathcal D\,\mathbf F^{-1}\mathrm{diag}(K_i)\,\mathbf F\,\mathrm{diag}(\theta_i)\,\vg,
  \qquad
  \mathbf A\Hm\sv=\sum_i \mathrm{diag}(\theta_i)\Hm\,\mathbf F\Hm\,\mathrm{diag}(K_i)\Hm\,(\mathbf F^{-1})\Hm\,\mathcal D\Hm\,\sv_i ,
  \label{eq:adjoint}
\end{equation}
with $\mathbf F\Hm=N\mathbf F^{-1}$, $(\mathbf F^{-1})\Hm=\mathbf F/N$ and $\mathcal D\Hm$ the insertion of $\Nrx-1$ zeros
between samples. The adjoint runs the chain backwards with conjugated factors and sums the channels (\cref{fig:afunctions}):
it focuses each channel, removes its relief phase and adds them, like a SAR processor.

Stored, the $12288\times12288$ complex $\mathbf A$ would take $12288^2\times\SI{16}{B}\approx\SI{2.4}{GB}$ and about
$10^8$ operations per product. Since every factor is diagonal or a fast transform, only the diagonals are kept
(\cref{tab:operators}): about \SI{1.6}{MB} and eight FFTs per product with four channels. \Cref{fig:axeschain} follows one
channel through the steps (axes: places $\ell$, Doppler bins $k$, samples $n$); \cref{fig:afill} shows what $\mathbf A$ would
contain; \cref{fig:afunctions} how it is used instead, as two functions.

Without relief the screens are constant and $\mathbf A$ splits by Doppler bin into $3072$ independent $4\times4$ blocks:
the standard reconstruction \eqref{eq:fold}. With relief the screen mixes the bins, so the whole line is solved at once,
iteratively.

\subsection{The problem that is solved}

\begin{equation}
  \hat{\vg}=\arg\min_{\vg}\ \|\mathbf A\vg-\sv\|^2+\mu^2\|\vg\|^2
  \quad\Longleftrightarrow\quad
  \bigl(\mathbf A\Hm\mathbf A+\mu^2\mathbf I\bigr)\,\hat{\vg}=\mathbf A\Hm\sv ,
  \qquad \mu=0.03\,\sigma_{\max}(\mathbf A).
  \label{eq:tikhonov}
\end{equation}
(zero gradient: $\mathbf A\Hm(\mathbf A\vg-\sv)+\mu^2\vg=\bm 0$). $\sigma_{\max}$ comes from 12 power iterations:
\[
  \bm x\leftarrow\mathbf A\Hm\mathbf A\bm x/\|\mathbf A\Hm\mathbf A\bm x\|,\qquad
  \sigma_{\max}\approx\sqrt{\|\mathbf A\Hm\mathbf A\bm x\|}.
\]

\paragraph{Why regularise.} With the singular values $\mathbf A\bm v_k=\sigma_k\bm u_k$, each direction $\bm v_k$ is
recovered with its own weight:
\begin{equation}
  \hat{\vg}=\sum_k \underbrace{\phi_k}_{\text{weight}}\,(\bm u_k\Hm\sv)\,\bm v_k ,
  \qquad
  \phi_k=\begin{cases}
    1/\sigma_k & \text{without $\mu$ (inverse)},\\[2pt]
    \dfrac{\sigma_k}{\sigma_k^2+\mu^2} & \text{with $\mu$ (Tikhonov)}.
  \end{cases}
  \label{eq:filterfactors}
\end{equation}
A small $\sigma_k$ is a direction the channels hardly measure, e.g.\ parcels seen with almost the same phases
(\cref{sec:mismatch}). There $1/\sigma_k$ would amplify noise and model errors (echo shape, height-table interpolation,
rounding). With $\mu$, $\phi_k\approx1/\sigma_k$ for $\sigma_k\gg\mu$ and $\phi_k\approx\sigma_k/\mu^2\to0$ for
$\sigma_k\ll\mu$. With $\mu=0.03\,\sigma_{\max}$ only directions measured below about \SI{3}{\percent} of the best are damped;
the second pass recovers part of what $\mu$ removed where the measurement is good.

\subsection{Conjugate gradients for least squares}

$\mathbf A\Hm\mathbf A+\mu^2\mathbf I$ is Hermitian positive definite, so CGLS solves \eqref{eq:tikhonov} with products by
$\mathbf A$ and $\mathbf A\Hm$ only, never forming the normal matrix (which would also square the condition number)
\cite{Hestenes1952,Trefethen1997} (\cref{fig:solveloop}):
\begin{itemize}
  \item each iteration: one model ($\bm q=\mathbf A\bm p$) and one adjoint ($\bm z=\mathbf A\Hm\bm r-\mu^2\bm\delta$);
        $\alpha=\|\bm z\|^2/(\|\bm q\|^2+\mu^2\|\bm p\|^2)$ is the optimal step and
        $\bm p\leftarrow\bm z+\beta\bm p$, $\beta=\|\bm z_{\mathrm{new}}\|^2/\|\bm z\|^2$, a direction conjugate to the
        previous ones;
  \item a pass stops at $\|\bm z\|\le10^{-4}\|\bm z_0\|$ or 60 iterations;
  \item the second pass solves for the residual of the first and adds it (iterated Tikhonov); the weight becomes
        $\bigl[1-\bigl(\mu^2/(\sigma_k^2+\mu^2)\bigr)^2\bigr]/\sigma_k$, closer to $1/\sigma_k$ where the channels measure
        well and still zero where they do not.
\end{itemize}
Every quantity is a vector or a number; $\mathbf A$ enters only through the two functions of \cref{fig:afunctions}.

\subsection{The answer: \texorpdfstring{$U=K\odot\mathbf F\hat{\vg}$}{U = K F g}}\label{sec:joint-answer}

\paragraph{What is wanted.} The line an ideal \emph{monostatic} radar at the transmitter would record at
$\PRF=\SI{2000}{Hz}$ without aliasing (the reference): the scene $\hat{\vg}$ recorded again by that radar.

\paragraph{Where it comes from.} \caseref{3}, \eqref{eq:case3}, applied to the scene found:
\begin{equation}
  U(f)=\underbrace{K(f)}_{\text{echo of one point}}\ \underbrace{\hat G(f)}_{\mathcal F\{\hat{\gs}\}}
  \quad\Longleftrightarrow\quad
  u[n]=\sum_\ell \hat{\gs}[\ell]\,k[n-\ell],
  \label{eq:answer}
\end{equation}
one chirp per position, weighted by $\hat{\gs}$.

\paragraph{Why without the screen.} The screen is the phase \emph{relative to the monostatic} path, so for the monostatic
radar it is one ($\bat=\bxt=0\Rightarrow\Delta C_0=0\Rightarrow\theta\equiv1$). The monostatic phase of each target is
common to all channels, part of $a_p$, so it is estimated in $\hat{\vg}$ and reaches $U$ as in the reference; the
channel-dependent phase, which creates the ambiguities, stays in $\mathbf A$.

\paragraph{Why back to the raw line, and not stop at $\hat{\vg}$.}
\begin{itemize}
  \item \emph{Fair comparison:} all methods return the raw line ($12288$ samples, \SI{2000}{Hz}) and are focused the same way.
  \item \emph{Band:} outside $|f|\le\Ba/2$ the data do not determine $\vg$; $K$ and the band cut discard it.
  \item \emph{SAR chain:} the reconstruction is a pre-processing step; the SAR processor expects raw data.
\end{itemize}
In practice $U=K\odot\mathbf F\hat{\vg}$, $U(f)=0$ for $|f|>\Ba/2$ ($\Ba=\SI{1234.8}{Hz}$), and the line is $\mathbf F^{-1}U$.

\paragraph{Afterwards (the same for every method).} The focusing is
$\text{image}=\mathbf F^{-1}\{K^*U\}=\mathbf F^{-1}\{|K|^2\hat G\}$: one peak per target ($\SI{-3}{dB}$ width
$\approx\SI{5.8}{m}$), scored with the metric of \cref{sec:metric} (\cref{fig:answer}).

\section{Properties}

\begin{itemize}
  \item \textbf{Consistency.} With a zero DEM the method reproduces the standard reconstruction within \SI{0.1}{\percent} rms.
  \item \textbf{Information used.} Only the channels, the geometry and the DEM; removing the target positions from the
        configuration does not change a single bit of the output.
  \item \textbf{DEM self-test.} The unexplained part of the channels is \SIrange{-41}{-39}{dB} with the right DEM and
        \SIrange{-20}{-7}{dB} with a wrong one (mirrored or shifted by \SI{1}{km}): an alarm for DEM errors.
  \item \textbf{Cost.} About \SI{3.3}{s} per line with four channels (median over 680 runs).
  \item \textbf{Stability.} Well-measured directions are recovered, poorly measured ones damped; three or more channels
        with unequally spaced cross-track baselines make almost every direction well measured (\cref{sec:mismatch}).
\end{itemize}
