\section{Signal model}

Point targets lie on a terrain profile $h(x)$, on the iso-range surface of the reference point ($x=0$, $h=0$, slant range
$r_0$), so only their height changes the bistatic geometry. Channel $i$ records
\begin{equation}
  s_i(t) = \sum_{p} a_p\, w(t-t_p)\, \exp\!\left(-\jj\frac{2\pi}{\lambda} \bigl[r_{\mathrm T,p}(t)+r_{\mathrm R,i,p}(t)\bigr]\right),
\end{equation}
with $a_p$ the complex amplitude of scatterer $p$ and $w$ the antenna pattern.

\section{Channel transfer functions and the coefficients \texorpdfstring{$C_0$, $C_1$, $C_2$, $\Delta t$}{C0, C1, C2, Delta t}}\label{sec:coeffs}

The reconstruction needs the transfer function $H_i(f)$ of each channel, the ratio of its Doppler spectrum to the monostatic
one for the same scatterer, derived from the range histories of the reference point \cite{Sakar2020tgrs,Sakar2022thesis}.

\paragraph{Step 1: two parabolas (time domain).} Let $M(t)=2r_{\mathrm T}(t)$ be the monostatic path and
$B_i(t)=r_{\mathrm T}(t)+r_{\mathrm R,i}(t)$ the bistatic path, both for the reference point. Over the aperture each is a
parabola:
\begin{equation}
  M(t)\approx M_{\min}+m_2\,(t-t_M)^2, \qquad B_i(t)\approx B_{i,\min}+b_{2,i}\,(t-t_{B,i})^2 ,
  \label{eq:parabolas}
\end{equation}
with minimum path lengths $M_{\min}$, $B_{i,\min}$ [m] at times $t_M$, $t_{B,i}$ [s] and curvatures $m_2$, $b_{2,i}$
[m/s$^2$].

\paragraph{Step 2: from time to Doppler (stationary phase).} For $\exp[-\jj2\pi P(t)/\lambda]$ with
$P(t)=P_{\min}+p_2(t-t_P)^2$, the instantaneous frequency is
\begin{equation}
  f(t)=-\frac{1}{\lambda}\frac{\mathrm dP}{\mathrm dt}=-\frac{2p_2}{\lambda}(t-t_P)
  \quad\Longrightarrow\quad t_f=t_P-\frac{\lambda f}{2p_2},
\end{equation}
so each Doppler frequency comes from one time $t_f$ (the ruler \eqref{eq:ruler}). By stationary phase, with
$P(t_f)=P_{\min}+\lambda^2f^2/(4p_2)$,
\begin{equation}
  \Psi_P(f) = -\frac{2\pi}{\lambda}P(t_f)-2\pi f t_f
  = -2\pi\Bigl[\frac{P_{\min}}{\lambda}+t_P\,f-\frac{\lambda f^2}{4p_2}\Bigr].
  \label{eq:posp}
\end{equation}
\paragraph{Step 3: the ratio of the two spectra (Doppler domain).} The amplitudes cancel; subtracting the phases
\eqref{eq:posp} of $B_i$ and $M$,
\begin{equation}
  H_i(f)=\e^{\jj[\Psi_{B_i}(f)-\Psi_M(f)]}
  =\exp\!\Bigl(-\jj2\pi\Bigl[\frac{C_{0,i}}{\lambda}+(\Delta t_i-C_{1,i})\,f+C_{2,i}\,\lambda f^2\Bigr]\Bigr),
  \label{eq:Hi}
\end{equation}
with
\begin{equation}
  C_{0,i}=B_{i,\min}-M_{\min},\qquad
  \Delta t_i-C_{1,i}=t_{B,i}-t_M,\qquad
  C_{2,i}=\frac{1}{4m_2}-\frac{1}{4b_{2,i}} .
  \label{eq:coeffs}
\end{equation}
In the simulator $\Delta t_i$ is the time between the sampled minima and $C_{1,i}$ the sub-sample correction from the fits.

$C_{0,i}$ [m] is a \emph{constant phase}, set by the cross-track baseline and the only coefficient that changes with
height (\cref{sec:dc0}); $\Delta t_i-C_{1,i}$ [s] a \emph{linear phase} (time shift, $\approx\bat_{,i}/(2v)$, the DPCA
sampling); $C_{2,i}$ [s$^2$/m] a small \emph{quadratic phase}. For the reference point (\cref{fig:coeffsref}):
$C_0\approx-\bxt\sin\theta$ ($\SI{-103}{m}$ for $\bxt=\SI{300}{m}$), independent of $\bat$; $\Delta t=\bat/(2v)$
(\SIlist{0.5;1.0;1.5}{ms}); $C_1<\SI{0.2}{\micro s}$ and $C_2<\SI{4e-7}{s^2/m}$, i.e.\ phases $2\pi C_1 f<\SI{1e-3}{rad}$
and $2\pi C_2\lambda f^2\approx\SI{0.2}{rad}$ at $f=\Ba/2$. The parabola fits the bistatic minus monostatic path (which
varies by \SI{5.6}{mm} over the aperture for $\bxt=\SI{300}{m}$) within $10^{-4}$\,\si{mm} (\cref{fig:rangefit}).

\section{The topographic residual \texorpdfstring{$\dC$}{Delta C0}}\label{sec:dc0}

For a scatterer at height $h$, $C_0$ differs from that of the reference point by the \emph{topographic residual}
\begin{equation}
  \Delta C_{0,i}(h) = C_{0,i}(\text{scatterer}) - C_{0,i}(\text{reference point}) \approx -\frac{\bxt_{,i}\, h}{r_0 \tan\theta},
  \qquad
  \varphi_i(h) = \frac{2\pi}{\lambda}\Delta C_{0,i}(h),
  \label{eq:dc0}
\end{equation}
the interferometric phase of the cross-track baseline. The simulator computes $\Delta C_{0,i}$ exactly; \eqref{eq:dc0}
agrees within \SI{1.8}{\percent} up to \SI{3}{km} (\cref{sec:dc0-proof}). At \SI{500}{m} the residuals are
\SIlist{0;-536;-214;-180}{mm}, two to four half-wavelengths. The other coefficients barely change (\cref{fig:coeffsres}): at
$f=\Ba/2$ and \SI{3}{km} with $\bxt=\SI{300}{m}$, $\Delta C_0$ gives about \SI{80}{rad}, $\Delta C_2$ \SI{4e-3}{rad},
$\Delta C_1$ \SI{4e-6}{rad}. On straight tracks topography is therefore a channel-dependent \emph{constant} phase per
scatterer; a baseline that changes during the aperture adds more (\cref{sec:timevar}).

\subsection{Derivation of the topographic residual}\label{sec:dc0-proof}

Flat earth; transmitter at $(vt,0,H)$, receiver $i$ at $(vt-\bat_{,i},\bxt_{,i},H)$ (azimuth, ground range, height). A
place of height $h$ on the iso-range surface is at $(x,y_h,h)$ with
\begin{equation}
  y_h^2+(H-h)^2=r_0^2 ,\qquad \cos\theta_h=\frac{H-h}{r_0},\qquad \sin\theta_h=\frac{y_h}{r_0},
  \label{eq:isorange}
\end{equation}
($\theta_0=\theta=\SI{20}{\degree}$).

\paragraph{Step 1: $C_0$ is a difference of closest-approach distances.} By \eqref{eq:coeffs},
$C_{0,i}=B_{i,\min}-M_{\min}$; the along-track offset only shifts the time of closest approach, so
\begin{equation}
  C_{0,i}\approx\bigl(r_{\mathrm T,\min}+r_{\mathrm R,i,\min}\bigr)-2r_{\mathrm T,\min}=r_{\mathrm R,i,\min}-r_{\mathrm T,\min},
\end{equation}
a one-way difference, with $r_{\mathrm T,\min}=r_0$ at every height on the iso-range.

\paragraph{Step 2: the receiver distance to first order in the baseline.} At closest approach,
\begin{align}
  r_{\mathrm R,i,\min}&=\sqrt{(y_h-\bxt_{,i})^2+(H-h)^2}
  =r_0\sqrt{1-\frac{2\bxt_{,i}\sin\theta_h}{r_0}+\frac{\bxt_{,i}^2}{r_0^2}}\notag\\
  &\approx r_0-\bxt_{,i}\sin\theta_h+\frac{\bxt_{,i}^2\cos^2\theta_h}{2r_0},
\end{align}
the first-order term being the baseline projected on the line of sight. Hence
\begin{equation}
  C_{0,i}(h)\approx-\bxt_{,i}\sin\theta_h+\frac{\bxt_{,i}^2\cos^2\theta_h}{2r_0}.
\end{equation}

\paragraph{Step 3: the look angle of a raised place.} From
\eqref{eq:isorange}, $\mathrm d(\cos\theta_h)=-\mathrm dh/r_0$, and
$\mathrm d(\sin\theta_h)=-\frac{\cos\theta_h}{\sin\theta_h}\,\mathrm d(\cos\theta_h)=\frac{\mathrm dh}{r_0\tan\theta_h}$, so
\begin{equation}
  \sin\theta_h-\sin\theta_0\approx\frac{h}{r_0\tan\theta_0}.
\end{equation}

\paragraph{Step 4: the residual.} Subtracting the reference point,
\begin{equation}
  \Delta C_{0,i}(h)=C_{0,i}(h)-C_{0,i}(0)\approx-\bxt_{,i}\bigl(\sin\theta_h-\sin\theta_0\bigr)\approx-\frac{\bxt_{,i}\,h}{r_0\tan\theta},
\end{equation}
i.e.\ \eqref{eq:dc0}, with $2\pi$ rather than $4\pi$ because only the receiver path changes. Geometrically
(\cref{fig:crosstrack}), the raised place moves by $s=h/\sin\theta$ perpendicular to the transmitter line of sight, and the
receiver line of sight is rotated by $\Delta\theta=\bxt_{,i}\cos\theta/r_0$, so its path changes by
$s\,\Delta\theta=\bxt_{,i}h/(r_0\tan\theta)$.

\paragraph{Validity.} Step~3 neglects
$\frac{1}{2}\,\mathrm d^2(\sin\theta)/\mathrm dh^2\,h^2$, a relative error of about $h/(2r_0\cos\theta\sin^2\theta)$:
\SI{0.6}{\percent} at \SI{1}{km} and \SI{1.8}{\percent} at \SI{3}{km}, with \eqref{eq:dc0} slightly too large. The term in
$\bxt^2$ is a fraction $\bxt\sin\theta/r_0\approx\num{1.3e-4}$ for \SI{300}{m}; on a curved Earth $\theta$ becomes the
local incidence angle. \Cref{tab:dc0check} compares both forms with the exact fits (\texttt{verify\_dC0.py} in
\texttt{runs/\allowbreak analysis\_v30/\allowbreak baselines\_solvers}). With $\bxt$ replaced by its change
$\varepsilon(t)$, the same steps give the phase of a time-varying baseline (\cref{sec:timevar}).
