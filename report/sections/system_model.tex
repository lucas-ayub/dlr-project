\section{Formation and parameters}

One platform transmits and receives (channel~0); the other $\Nrx-1$ receivers fly behind it, displaced by $\bat_{,i}$
along track and $\bxt_{,i}$ across track (horizontally, towards the scene) (\cref{fig:geometry}). The along-track baselines

(a random draw scaled to a maximum of \SI{300}{m}); channel~0 is the transmitter ($\bat=\bxt=0$). Parameters:

\section{Signal model}

Point targets lie on a terrain profile $h(x)$ along azimuth, on the iso-range surface of the reference point ($x=0$, $h=0$,
slant range $r_0$), so only their height changes the bistatic geometry. The azimuth signal of channel $i$ is
\begin{equation}
  s_i(t) = \sum_{p} a_p\, w(t-t_p)\, \exp\!\left(-\jj\frac{2\pi}{\lambda} \bigl[r_{\mathrm T,p}(t)+r_{\mathrm R,i,p}(t)\bigr]\right),
\end{equation}
with $a_p$ the complex amplitude of scatterer $p$, $w$ the antenna pattern and $r_{\mathrm T}+r_{\mathrm R,i}$ the bistatic
path (\cref{fig:geometry}).

\section{Channel transfer functions and the coefficients \texorpdfstring{$C_0$, $C_1$, $C_2$, $\Delta t$}{C0, C1, C2, Delta t}}\label{sec:coeffs}

The reconstruction (\cref{ch:recon}) needs the transfer function $H_i(f)$ of each channel: the ratio between its Doppler
spectrum and the monostatic one for the same scatterer. We derive it from the range histories of the reference point,
following \cite{Sakar2020tgrs,Sakar2022thesis}.

\paragraph{Step 1: two parabolas (time domain).} Let $M(t)=2r_{\mathrm T}(t)$ be the monostatic path and
$B_i(t)=r_{\mathrm T}(t)+r_{\mathrm R,i}(t)$ the bistatic path, both for the reference point. Over the aperture each is a
parabola around its own minimum:
\begin{equation}
  M(t)\approx M_{\min}+m_2\,(t-t_M)^2, \qquad B_i(t)\approx B_{i,\min}+b_{2,i}\,(t-t_{B,i})^2 ,
  \label{eq:parabolas}
\end{equation}
where $M_{\min}$, $B_{i,\min}$ are the smallest path lengths [m], $t_M$, $t_{B,i}$ the times at which they occur [s] and
$m_2$, $b_{2,i}$ the curvatures [m/s$^2$]. The corresponding azimuth signals are
$s_M(t)=w(t)\exp[-\jj 2\pi M(t)/\lambda]$ and $s_i(t)=w(t)\exp[-\jj 2\pi B_i(t)/\lambda]$.

\paragraph{Step 2: from time to Doppler (stationary phase).} For a signal $\exp[-\jj2\pi P(t)/\lambda]$ with a parabolic path
$P(t)=P_{\min}+p_2(t-t_P)^2$, the instantaneous frequency is
\begin{equation}
  f(t)=-\frac{1}{\lambda}\frac{\mathrm dP}{\mathrm dt}=-\frac{2p_2}{\lambda}(t-t_P)
  \quad\Longrightarrow\quad t_f=t_P-\frac{\lambda f}{2p_2},
\end{equation}
so each Doppler frequency $f$ comes from one time $t_f$ (the ruler \eqref{eq:ruler} of \cref{ch:recon}). By stationary
phase the spectral phase at $f$ is the signal phase at $t_f$ minus $2\pi f t_f$. With
$P(t_f)=P_{\min}+\lambda^2f^2/(4p_2)$,
\begin{equation}
  \Psi_P(f) = -\frac{2\pi}{\lambda}P(t_f)-2\pi f t_f
  = -2\pi\Bigl[\frac{P_{\min}}{\lambda}+t_P\,f-\frac{\lambda f^2}{4p_2}\Bigr].
  \label{eq:posp}
\end{equation}
The spectral amplitude (antenna pattern) is the same for both paths.

\paragraph{Step 3: the ratio of the two spectra (Doppler domain).} Applying \eqref{eq:posp} to $B_i$ and $M$ and
subtracting the phases:
\begin{equation}
  H_i(f)=\e^{\jj[\Psi_{B_i}(f)-\Psi_M(f)]}
  =\exp\!\Bigl(-\jj2\pi\Bigl[\frac{C_{0,i}}{\lambda}+(\Delta t_i-C_{1,i})\,f+C_{2,i}\,\lambda f^2\Bigr]\Bigr),
  \label{eq:Hi}
\end{equation}
with
\begin{equation}
  C_{0,i}=B_{i,\min}-M_{\min},\qquad
  \Delta t_i-C_{1,i}=t_{B,i}-t_M,\qquad
  C_{2,i}=\frac{1}{4m_2}-\frac{1}{4b_{2,i}} .
  \label{eq:coeffs}
\end{equation}
In the simulator $\Delta t_i$ is the time between the samples where $M$ and $B_i$ are smallest and $C_{1,i}$ the sub-sample
correction from the fitted parabolas, so $\Delta t_i-C_{1,i}$ is the time between the two minima.

\paragraph{Physical meaning.}
\begin{itemize}
  \item $C_{0,i}$ [m], the difference of the shortest paths: a \emph{constant phase} $2\pi C_{0,i}/\lambda$, set by the
        cross-track baseline; the coefficient that changes with height (\cref{sec:dc0}).
  \item $\Delta t_i-C_{1,i}$ [s], the delay between the minima: a \emph{linear phase} (time shift), set by the along-track
        baseline ($\approx\bat_{,i}/(2v)$); it gives the DPCA sampling.
  \item $C_{2,i}$ [s$^2$/m], the difference of inverse curvatures: a \emph{quadratic phase}, a small residual chirp.
\end{itemize}
For another scatterer the same derivation holds with its own paths; to first order only $C_{0,i}$ changes (\cref{sec:dc0}).

For the reference point (\cref{fig:coeffsref}): $C_0\approx-\bxt\sin\theta$ ($\SI{-171}{m}$ for $\bxt=\SI{500}{m}$),
independent of $\bat$; $\Delta t=\bat/(2v)$ (\SIlist{0.5;1.0;1.5}{ms}), independent of $\bxt$; $C_1<\SI{0.2}{\micro s}$
and $C_2<\SI{4e-7}{s^2/m}$, i.e.\ phases $2\pi C_1 f<\SI{1e-3}{rad}$ and $2\pi C_2\lambda f^2\approx\SI{0.2}{rad}$ at
$f=\Ba/2$. The parabola \eqref{eq:parabolas} fits the bistatic minus monostatic path, which varies by \SI{5.6}{mm} over the
aperture for $\bxt=\SI{300}{m}$, within $10^{-4}$\,\si{mm} (\cref{fig:rangefit}).

\section{The topographic residual \texorpdfstring{$\dC$}{Delta C0}}\label{sec:dc0}

For a scatterer at height $h$, $C_0$ differs from that of the reference point by the \emph{topographic residual}
\begin{equation}
  \Delta C_{0,i}(h) = C_{0,i}(\text{scatterer}) - C_{0,i}(\text{reference point}) \approx -\frac{\bxt_{,i}\, h}{r_0 \tan\theta},
  \qquad
  \varphi_i(h) = \frac{2\pi}{\lambda}\Delta C_{0,i}(h),
  \label{eq:dc0}
\end{equation}
the interferometric phase of the cross-track baseline (zero for the transmitter channel, $\bxt=0$). The simulator computes
$\Delta C_{0,i}$ exactly; \eqref{eq:dc0} agrees within \SI{1.8}{\percent} up to \SI{3}{km} (\cref{sec:dc0-proof}). At
\SI{500}{m} the residuals are $[0,-536,-214,-180]$\,\si{mm}, two to four half-wavelengths. The other coefficients barely
change (\cref{fig:coeffsres}): at $f=\Ba/2$ and \SI{3}{km} with $\bxt=\SI{300}{m}$, $\Delta C_0$ gives about \SI{80}{rad},
$\Delta C_2$ \SI{4e-3}{rad} and $\Delta C_1$ \SI{4e-6}{rad}; $\Delta t$ does not change. Topography therefore acts as a
channel-dependent \emph{constant} phase per scatterer, which the joint reconstruction uses (\cref{ch:joint}). This holds for
straight tracks; a baseline that changes during the aperture adds a phase $\Delta C_0$ does not describe (\cref{sec:timevar}).

\subsection{Derivation of the topographic residual}\label{sec:dc0-proof}

Flat earth, azimuth $x$, ground range $y$ (towards the scene), height $z$. The transmitter flies at
$(vt,0,H)$ and receiver $i$ at $(vt-\bat_{,i},\bxt_{,i},H)$. A place of height $h$ on the iso-range surface of the reference
point is at $(x,y_h,h)$ with
\begin{equation}
  y_h^2+(H-h)^2=r_0^2 ,\qquad \cos\theta_h=\frac{H-h}{r_0},\qquad \sin\theta_h=\frac{y_h}{r_0},
  \label{eq:isorange}
\end{equation}
with $\theta_h$ the transmitter look angle ($\theta_0=\theta=\SI{20}{\degree}$).

\paragraph{Step 1: $C_0$ is a difference of closest-approach distances.} By \eqref{eq:coeffs},
$C_{0,i}=B_{i,\min}-M_{\min}$. The along-track offset only shifts the time of closest approach ($\Delta t_i$), not the
distance, so to first order
\begin{equation}
  C_{0,i}\approx\bigl(r_{\mathrm T,\min}+r_{\mathrm R,i,\min}\bigr)-2r_{\mathrm T,\min}=r_{\mathrm R,i,\min}-r_{\mathrm T,\min},
\end{equation}
a one-way difference, with $r_{\mathrm T,\min}=r_0$ at every height on the iso-range.

\paragraph{Step 2: the receiver distance to first order in the baseline.} At closest approach,
\begin{align}
  r_{\mathrm R,i,\min}&=\sqrt{(y_h-\bxt_{,i})^2+(H-h)^2}
  =r_0\sqrt{1-\frac{2\bxt_{,i}\sin\theta_h}{r_0}+\frac{\bxt_{,i}^2}{r_0^2}}\notag\\
  &\approx r_0-\bxt_{,i}\sin\theta_h+\frac{\bxt_{,i}^2\cos^2\theta_h}{2r_0},
\end{align}
using \eqref{eq:isorange} and $\sqrt{1+\delta}\approx1+\delta/2-\delta^2/8$. The first-order term is the baseline
projected on the line of sight, $\partial r_{\mathrm R}/\partial\bxt\approx-\sin\theta_h$. Hence
\begin{equation}
  C_{0,i}(h)\approx-\bxt_{,i}\sin\theta_h+\frac{\bxt_{,i}^2\cos^2\theta_h}{2r_0}.
\end{equation}
(at $h=0$, the $C_0\approx-\bxt\sin\theta$ of \cref{fig:coeffsref}).

\paragraph{Step 3: the look angle of a raised place.} From
\eqref{eq:isorange}, $\mathrm d(\cos\theta_h)=-\mathrm dh/r_0$, and
$\mathrm d(\sin\theta_h)=-\frac{\cos\theta_h}{\sin\theta_h}\,\mathrm d(\cos\theta_h)=\frac{\mathrm dh}{r_0\tan\theta_h}$, so
\begin{equation}
  \sin\theta_h-\sin\theta_0\approx\frac{h}{r_0\tan\theta_0}.
\end{equation}

\paragraph{Step 4: the residual.} Subtracting the reference point,
\begin{equation}
  \Delta C_{0,i}(h)=C_{0,i}(h)-C_{0,i}(0)\approx-\bxt_{,i}\bigl(\sin\theta_h-\sin\theta_0\bigr)\approx-\frac{\bxt_{,i}\,h}{r_0\tan\theta},
\end{equation}
with $\varphi_i(h)=2\pi\Delta C_{0,i}(h)/\lambda$, i.e.\ \eqref{eq:dc0}. The factor is $2\pi$, not $4\pi$ as in a
monostatic interferometer, because only the receiver path changes. Geometrically (\cref{fig:crosstrack}), the raised place
moves by $s=h/\sin\theta$ perpendicular to the transmitter line of sight, while the receiver line of sight is rotated by
$\Delta\theta=\bxt_{,i}\cos\theta/r_0$, so its path changes by $s\,\Delta\theta=\bxt_{,i}h/(r_0\tan\theta)$.

\paragraph{Validity.} Step~3 neglects
$\frac{1}{2}\,\mathrm d^2(\sin\theta)/\mathrm dh^2\,h^2$, a relative error of about $h/(2r_0\cos\theta\sin^2\theta)$:
\SI{0.6}{\percent} at \SI{1}{km} and \SI{1.8}{\percent} at \SI{3}{km}, with \eqref{eq:dc0} slightly too large. The term in
$\bxt^2$ is a fraction $\bxt\sin\theta/r_0\approx\num{1.3e-4}$ for \SI{300}{m}. On a curved Earth the look angle becomes
the local incidence angle. The same steps, with $\bxt$ replaced by its change $\varepsilon(t)$, give the phase of a
time-varying baseline (\cref{sec:timevar}).
