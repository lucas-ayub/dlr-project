% DEM in radar geometry: the same slant range r meets the terrain at another ground range than the flat terrain.
% Platform P = (0, 4.6); terrain h(y) = 1.6 exp(-((y-6)/1.2)^2); flat point F = (5, 0); r = |PF| = 6.794;
% terrain point T = (6.091, 1.591) solves y^2 + (4.6 - h(y))^2 = r^2 (computed once, see the caption).
\begin{tikzpicture}[x=1cm,y=1cm]
  \coordinate (P) at (0,4.6);
  \coordinate (F) at (5.0,0);
  \coordinate (T) at (6.091,1.591);
  % terrain and flat reference
  \fill[terrainfill] (0.6,0) -- plot[smooth,domain=0.6:9.4,samples=80] (\x,{1.6*exp(-((\x-6.0)/1.2)^2)}) -- (9.4,-0.25) -- (0.6,-0.25) -- cycle;
  \draw[terrain,line width=0.9pt] plot[smooth,domain=0.6:9.4,samples=80] (\x,{1.6*exp(-((\x-6.0)/1.2)^2)});
  \draw[inkb,dashed] (0.6,0) -- (9.4,0) node[right,nota]{flat terrain};
  % platform and slant range
  \node[sat] at (P) {};
  \node[nota,anchor=south] at ($(P)+(0,0.15)$) {platform (track into the page)};
  \draw[cblue,line width=0.8pt] (P) -- (T) node[pos=0.45,above,sloped,nota,text=cblue]{slant range $r$};
  \draw[cblue!60,line width=0.5pt,densely dotted] (P) -- (F);
  % arc of constant range through the flat point and the terrain point
  \draw[cblue,line width=0.9pt] (P) ++(-44:6.794) arc[start angle=-44,end angle=-22,radius=6.794];
  \node[nota,text=cblue,anchor=west,align=left] at ($(P)+(-22:6.794)+(0.08,0.02)$) {same range $r$};
  % the two points
  \fill[inkb] (F) circle (2pt);
  \fill[cblue] (T) circle (2.2pt);
  % height of the terrain point
  \draw[cota] (6.091,0) -- node[right,nota,text=ink]{$h$} (6.091,1.5);
  % ground-range shift
  \draw[inkb,densely dotted] (F) -- (5.0,-0.55);
  \draw[inkb,densely dotted] (6.091,0) -- (6.091,-0.55);
  \draw[cota] (5.0,-0.5) -- (6.091,-0.5);
  \node[nota,anchor=north] at (5.55,-0.55) {same $r$, other $y$};
  \draw[->,inkb] (0.6,-1.25) -- (9.4,-1.25) node[right,nota]{ground range $y$};
\end{tikzpicture}
